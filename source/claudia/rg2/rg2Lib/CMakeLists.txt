cmake_minimum_required(VERSION 3.15)

project(rg2Lib)

# A track adds its own source names to its own per-track list below and to
# nothing else.

add_library(rg2Lib STATIC)

# ----------------- the per-track source lists
#
# Each list appends to RG2LIB_SOURCES and is owned by the track its name
# carries.

set(RG2LIB_SOURCES)

include(sources_board.cmake)
include(sources_sched.cmake)
include(sources_chain.cmake)
include(sources_dsp.cmake)
include(sources_repo.cmake)
include(sources_plugin.cmake)
include(sources_proto.cmake)
include(sources_oracle.cmake)
include(sources_perf.cmake)
include(sources_int.cmake)

# A CMake target with no source at all fails at generate time. While every list
# is still empty the target carries one translation unit GENERATED INTO THE
# BUILD TREE, not a committed source file. The placeholder disappears by itself
# as soon as the first track adds a name to its own list.
if(NOT RG2LIB_SOURCES)
	set(RG2LIB_PLACEHOLDER ${CMAKE_CURRENT_BINARY_DIR}/rg2LibPlaceholder.cpp)
	file(WRITE ${RG2LIB_PLACEHOLDER}
		"// Generated by rg2Lib/CMakeLists.txt while every per-track source list is empty.\n"
		"namespace rg2 { extern const int rg2LibPlaceholder; const int rg2LibPlaceholder = 0; }\n")
	list(APPEND RG2LIB_SOURCES ${RG2LIB_PLACEHOLDER})
endif()

target_sources(rg2Lib PRIVATE ${RG2LIB_SOURCES})
source_group("source" FILES ${RG2LIB_SOURCES})

# ----------------- the link line
#
# hardwareLib is the right layer and it carries what the G2 needs: am29f for the
# flash, i2c and i2cFlash, lcd and lcdfonts for the panel display, and sciMidi.
#
# wLib is the Waldorf layer and this target does NOT link it. See
# docs/divergence.md for the reason and the condition that would re-open it.

target_link_libraries(rg2Lib PUBLIC hardwareLib dsp56kEmu synthLib)

if(DSP56300_DEBUGGER)
	target_link_libraries(rg2Lib PUBLIC dsp56kDebugger)
endif()

# gdbStub.cpp's loopback socket. On Windows the Berkeley calls live in ws2_32
# and nothing else in this library pulls it in; every other platform carries
# them in libc and needs no entry here.
if(WIN32)
	target_link_libraries(rg2Lib PUBLIC ws2_32)
endif()

set_property(TARGET rg2Lib PROPERTY FOLDER "RG2")

target_include_directories(rg2Lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_include_directories(rg2Lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# ----------------- the ColdFire core
#
# The link is BEHIND AN OPTION and the option defaults to ON, because the target
# coldfire::coldfire now exists. The option stays, rather than becoming an
# unconditional link, so a build that only wants the DSP side of rg2Lib can turn
# the core off and skip fetching a second repository.
# The root CMakeLists.txt declares where the sources come from.

option(RG2_LINK_COLDFIRE "Link the ColdFire core" ON)

if(RG2_LINK_COLDFIRE)
	if(NOT TARGET coldfire::coldfire)
		if(RG2_COLDFIRE_SOURCE_DIR)
			option(RG2_COLDFIRE_SOURCE_DIR_IS_NOT_THE_PIN "Build from RG2_COLDFIRE_SOURCE_DIR even when that checkout is not the pinned commit" OFF)
			find_program(G2_GIT_EXECUTABLE NAMES git)

			set(coldfireHeadIsPin FALSE)
			set(coldfireDivergence "")
			execute_process(
				COMMAND ${G2_GIT_EXECUTABLE} -C "${RG2_COLDFIRE_SOURCE_DIR}" rev-parse HEAD
				OUTPUT_VARIABLE coldfireHead OUTPUT_STRIP_TRAILING_WHITESPACE
				ERROR_QUIET RESULT_VARIABLE coldfireHeadResult)
			execute_process(
				COMMAND ${G2_GIT_EXECUTABLE} -C "${RG2_COLDFIRE_SOURCE_DIR}" rev-parse "${RG2_COLDFIRE_GIT_TAG}^{commit}"
				OUTPUT_VARIABLE coldfirePin OUTPUT_STRIP_TRAILING_WHITESPACE
				ERROR_QUIET RESULT_VARIABLE coldfirePinResult)
			execute_process(
				COMMAND ${G2_GIT_EXECUTABLE} -C "${RG2_COLDFIRE_SOURCE_DIR}" status --porcelain
				OUTPUT_VARIABLE coldfireDirty OUTPUT_STRIP_TRAILING_WHITESPACE
				ERROR_QUIET RESULT_VARIABLE coldfireDirtyResult)

			if(NOT coldfireHeadResult EQUAL 0 OR NOT coldfireDirtyResult EQUAL 0)
				set(coldfireDivergence "its state could not be read: `git -C <dir> rev-parse HEAD` or `git status` failed, so this tree cannot be compared against the pin at all")
			elseif(NOT coldfirePinResult EQUAL 0)
				set(coldfireDivergence "it does not contain the pinned commit ${RG2_COLDFIRE_GIT_TAG} at all; its HEAD is ${coldfireHead}")
			elseif(NOT coldfireHead STREQUAL coldfirePin)
				set(coldfireDivergence "its HEAD is ${coldfireHead}, and the pin is ${coldfirePin}")
			elseif(NOT coldfireDirty STREQUAL "")
				set(coldfireDivergence "its working tree has uncommitted changes, so what would compile is not the pinned commit even though its HEAD is")
			else()
				set(coldfireHeadIsPin TRUE)
			endif()

			if(NOT coldfireHeadIsPin AND NOT RG2_COLDFIRE_SOURCE_DIR_IS_NOT_THE_PIN)
				message(FATAL_ERROR
					"RG2_COLDFIRE_SOURCE_DIR points at ${RG2_COLDFIRE_SOURCE_DIR}, but "
					"${coldfireDivergence}. Every result from this build tree would be a "
					"statement about that tree and not about the pin.\n"
					"Unset RG2_COLDFIRE_SOURCE_DIR to build against the pin, check the "
					"sibling out onto ${RG2_COLDFIRE_GIT_TAG}, or pass "
					"-DRG2_COLDFIRE_SOURCE_DIR_IS_NOT_THE_PIN=ON to build against it "
					"deliberately — and then label every result from this build tree as "
					"not testing the pin.")
			endif()
			if(NOT coldfireHeadIsPin)
				message(WARNING
					"coldfire: building against ${RG2_COLDFIRE_SOURCE_DIR}, not the pin, because "
					"${coldfireDivergence}. Results from this build tree do not describe the pin.")
			endif()

			add_subdirectory(${RG2_COLDFIRE_SOURCE_DIR} ${CMAKE_BINARY_DIR}/coldfire EXCLUDE_FROM_ALL)
		else()
			include(FetchContent)
			FetchContent_MakeAvailable(coldfire)
		endif()
	endif()
	target_link_libraries(rg2Lib PUBLIC coldfire::coldfire)
endif()
