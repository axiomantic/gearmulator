cmake_minimum_required(VERSION 3.15)

project(rg2TestConsole)

# This directory is shared by order and not by ownership.

add_executable(rg2TestConsole)

set(SOURCES)

if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp)
	list(APPEND SOURCES main.cpp)
endif()

# An executable with no source at all cannot link. Until main.cpp exists, the
# target carries one translation unit generated into the build tree, not a
# committed source file. The placeholder disappears by itself as soon as
# main.cpp exists.
if(NOT SOURCES)
	set(RG2_TEST_CONSOLE_PLACEHOLDER ${CMAKE_CURRENT_BINARY_DIR}/rg2TestConsolePlaceholder.cpp)
	file(WRITE ${RG2_TEST_CONSOLE_PLACEHOLDER}
		"// Generated by rg2TestConsole/CMakeLists.txt until main.cpp exists.\n"
		"int main(int, char**) { return 0; }\n")
	list(APPEND SOURCES ${RG2_TEST_CONSOLE_PLACEHOLDER})
endif()

target_sources(rg2TestConsole PRIVATE ${SOURCES})
source_group("source" FILES ${SOURCES})

target_link_libraries(rg2TestConsole PUBLIC rg2Lib)

set_property(TARGET rg2TestConsole PROPERTY FOLDER "RG2")
