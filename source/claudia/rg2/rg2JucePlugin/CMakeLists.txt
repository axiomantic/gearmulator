cmake_minimum_required(VERSION 3.15)

project(rg2JucePlugin VERSION ${CMAKE_PROJECT_VERSION})

# The target is not yet a JUCE plugin. The createJucePlugin() macro that
# every other product in this tree uses takes a product name, a
# four-character plugin code, a binary-data target and a skin, and none of
# those four values is stated anywhere.

add_library(rg2JucePlugin STATIC)

set(SOURCES)

# The synthLib::Device subclass. The
# tests t0_device_surface and t0_handoff_flags compile this file DIRECTLY (the
# rg2PatchLoad.cpp arrangement), so this target's copy and theirs are one
# translation unit, not two.
list(APPEND SOURCES rg2Device.cpp)

# The state format the Device's overrides delegate to.
# The same tests compile it directly alongside rg2Device.cpp, for the same
# reason.
list(APPEND SOURCES rg2State.cpp)

# The synthLib::Plugin subclass that sets the resampler mode.
# t0_resampler_mode compiles it directly, the same
# arrangement.
list(APPEND SOURCES rg2Plugin.cpp)

# The plugin track appends its own sources here. Until it does, the target
# carries one translation unit GENERATED INTO THE BUILD TREE, for the same
# reason rg2Lib does.
if(NOT SOURCES)
	set(G2_JUCE_PLUGIN_PLACEHOLDER ${CMAKE_CURRENT_BINARY_DIR}/rg2JucePluginPlaceholder.cpp)
	file(WRITE ${G2_JUCE_PLUGIN_PLACEHOLDER}
		"// Generated by rg2JucePlugin/CMakeLists.txt until the plugin track adds a source.\n"
		"namespace rg2 { extern const int rg2JucePluginPlaceholder; const int rg2JucePluginPlaceholder = 0; }\n")
	list(APPEND SOURCES ${G2_JUCE_PLUGIN_PLACEHOLDER})
endif()

target_sources(rg2JucePlugin PRIVATE ${SOURCES})
source_group("source" FILES ${SOURCES})

target_link_libraries(rg2JucePlugin PUBLIC rg2Lib)

set_property(TARGET rg2JucePlugin PROPERTY FOLDER "RG2")
