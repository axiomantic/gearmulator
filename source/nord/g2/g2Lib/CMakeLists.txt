cmake_minimum_required(VERSION 3.15)

project(g2Lib)

# A track adds its own source names to its own per-track list below and to
# nothing else.

add_library(g2Lib STATIC)

# ----------------- the per-track source lists
#
# Each list appends to G2LIB_SOURCES and is owned by the track its name
# carries.

set(G2LIB_SOURCES)

include(sources_board.cmake)
include(sources_sched.cmake)
include(sources_chain.cmake)
include(sources_dsp.cmake)
include(sources_repo.cmake)
include(sources_plugin.cmake)
include(sources_proto.cmake)
include(sources_oracle.cmake)
include(sources_perf.cmake)
include(sources_int.cmake)

# A CMake target with no source at all fails at generate time. While every list
# is still empty the target carries one translation unit GENERATED INTO THE
# BUILD TREE, not a committed source file. The placeholder disappears by itself
# as soon as the first track adds a name to its own list.
if(NOT G2LIB_SOURCES)
	set(G2LIB_PLACEHOLDER ${CMAKE_CURRENT_BINARY_DIR}/g2LibPlaceholder.cpp)
	file(WRITE ${G2LIB_PLACEHOLDER}
		"// Generated by g2Lib/CMakeLists.txt while every per-track source list is empty.\n"
		"namespace g2 { extern const int g2LibPlaceholder; const int g2LibPlaceholder = 0; }\n")
	list(APPEND G2LIB_SOURCES ${G2LIB_PLACEHOLDER})
endif()

target_sources(g2Lib PRIVATE ${G2LIB_SOURCES})
source_group("source" FILES ${G2LIB_SOURCES})

# ----------------- the link line
#
# hardwareLib is the right layer and it carries what the G2 needs: am29f for the
# flash, i2c and i2cFlash, lcd and lcdfonts for the panel display, and sciMidi.
#
# wLib is the Waldorf layer and this target does NOT link it. See
# docs/divergence.md for the reason and the condition that would re-open it.

target_link_libraries(g2Lib PUBLIC hardwareLib dsp56kEmu synthLib)

if(DSP56300_DEBUGGER)
	target_link_libraries(g2Lib PUBLIC dsp56kDebugger)
endif()

# gdbStub.cpp's loopback socket. On Windows the Berkeley calls live in ws2_32
# and nothing else in this library pulls it in; every other platform carries
# them in libc and needs no entry here.
if(WIN32)
	target_link_libraries(g2Lib PUBLIC ws2_32)
endif()

set_property(TARGET g2Lib PROPERTY FOLDER "G2")

target_include_directories(g2Lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_include_directories(g2Lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# ----------------- the MCF5407 core
#
# The link is BEHIND AN OPTION and the option defaults to ON, because the target
# mcf5407::mcf5407 now exists. The option stays, rather than becoming an
# unconditional link, so a build that only wants the DSP side of g2Lib can turn
# the core off and skip fetching a second repository.
# The root CMakeLists.txt declares where the sources come from.

option(G2_LINK_MCF5407 "Link the MCF5407 core" ON)

if(G2_LINK_MCF5407)
	if(G2_MCF5407_SOURCE_DIR)
		# A sibling checkout sits wherever its last piece of work left it, which is
		# usually neither the pin nor whatever the pin is being moved to, so an
		# unannounced override quietly makes every result a statement about a third
		# tree. A diverged tree is not itself wrong — iterating on both
		# repositories at once is the reason this override exists — so what is
		# fatal is entering that state without saying so. A warning alone would
		# not do: an unread line in a configure log is the signal this whole
		# failure class already gets past. The acknowledgement is the label, and
		# once given, the divergence is restated on every configure that follows.
		option(G2_MCF5407_SOURCE_DIR_IS_NOT_THE_PIN "Build from G2_MCF5407_SOURCE_DIR even when that checkout is not the pinned commit" OFF)
		find_program(G2_GIT_EXECUTABLE NAMES git)

		set(mcf5407HeadIsPin FALSE)
		set(mcf5407Divergence "")
		execute_process(
			COMMAND ${G2_GIT_EXECUTABLE} -C "${G2_MCF5407_SOURCE_DIR}" rev-parse HEAD
			OUTPUT_VARIABLE mcf5407Head OUTPUT_STRIP_TRAILING_WHITESPACE
			ERROR_QUIET RESULT_VARIABLE mcf5407HeadResult)
		execute_process(
			COMMAND ${G2_GIT_EXECUTABLE} -C "${G2_MCF5407_SOURCE_DIR}" rev-parse "${G2_MCF5407_GIT_TAG}^{commit}"
			OUTPUT_VARIABLE mcf5407Pin OUTPUT_STRIP_TRAILING_WHITESPACE
			ERROR_QUIET RESULT_VARIABLE mcf5407PinResult)
		execute_process(
			COMMAND ${G2_GIT_EXECUTABLE} -C "${G2_MCF5407_SOURCE_DIR}" status --porcelain
			OUTPUT_VARIABLE mcf5407Dirty OUTPUT_STRIP_TRAILING_WHITESPACE
			ERROR_QUIET RESULT_VARIABLE mcf5407DirtyResult)

		if(NOT mcf5407HeadResult EQUAL 0 OR NOT mcf5407DirtyResult EQUAL 0)
			set(mcf5407Divergence "its state could not be read: `git -C <dir> rev-parse HEAD` or `git status` failed, so this tree cannot be compared against the pin at all")
		elseif(NOT mcf5407PinResult EQUAL 0)
			set(mcf5407Divergence "it does not contain the pinned commit ${G2_MCF5407_GIT_TAG} at all; its HEAD is ${mcf5407Head}")
		elseif(NOT mcf5407Head STREQUAL mcf5407Pin)
			set(mcf5407Divergence "its HEAD is ${mcf5407Head}, and the pin is ${mcf5407Pin}")
		elseif(NOT mcf5407Dirty STREQUAL "")
			set(mcf5407Divergence "its working tree has uncommitted changes, so what would compile is not the pinned commit even though its HEAD is")
		else()
			set(mcf5407HeadIsPin TRUE)
		endif()

		if(NOT mcf5407HeadIsPin AND NOT G2_MCF5407_SOURCE_DIR_IS_NOT_THE_PIN)
			message(FATAL_ERROR
				"G2_MCF5407_SOURCE_DIR points at ${G2_MCF5407_SOURCE_DIR}, but "
				"${mcf5407Divergence}. Every result from this build tree would be a "
				"statement about that tree and not about the pin.\n"
				"Unset G2_MCF5407_SOURCE_DIR to build against the pin, check the "
				"sibling out onto ${G2_MCF5407_GIT_TAG}, or pass "
				"-DG2_MCF5407_SOURCE_DIR_IS_NOT_THE_PIN=ON to build against it "
				"deliberately — and then label every result from this build tree as "
				"not testing the pin.")
		endif()
		if(NOT mcf5407HeadIsPin)
			message(WARNING
				"mcf5407: building against ${G2_MCF5407_SOURCE_DIR}, not the pin, because "
				"${mcf5407Divergence}. Results from this build tree do not describe the pin.")
		endif()

		add_subdirectory(${G2_MCF5407_SOURCE_DIR} ${CMAKE_BINARY_DIR}/mcf5407 EXCLUDE_FROM_ALL)
	else()
		include(FetchContent)
		FetchContent_MakeAvailable(mcf5407)
	endif()
	target_link_libraries(g2Lib PUBLIC mcf5407::mcf5407)
endif()

# ----------------- the configure-time guard of the two UNMEASURED bus symbols
#
# G2_MCU_BUS_CLOCK_HZ and G2_MCU_BUS_DIVIDER are UNMEASURED and both read 0u
# today. A source that uses either one computes with
# a zero, so the build fails at configure time and names the symbol.
#
# The one file allowed to carry the two names is their declaration site,
# g2Lib/g2/timebase.h.

set(G2_UNMEASURED_SYMBOLS G2_MCU_BUS_CLOCK_HZ G2_MCU_BUS_DIVIDER)

# BOTH SIDES OF THE COMPARISON ARE NORMALISED, and that is load-bearing.
# file(GLOB_RECURSE) returns each path with the glob pattern's own spelling
# preserved, so a pattern that carries ".." yields ".../g2Lib/../g2Lib/g2/
# timebase.h". That string never compares equal to the declaration site written
# without the "..", the exemption below never fires for any file, and the guard
# then rejects its own declaration site. REALPATH collapses both spellings.
get_filename_component(G2_UNMEASURED_DECLARATION_SITE
	"${CMAKE_CURRENT_SOURCE_DIR}/g2/timebase.h" REALPATH)

file(GLOB_RECURSE G2_GUARDED_FILES
	${CMAKE_CURRENT_SOURCE_DIR}/../*.c
	${CMAKE_CURRENT_SOURCE_DIR}/../*.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/../*.h
	${CMAKE_CURRENT_SOURCE_DIR}/../*.hpp)

foreach(guardedFile IN LISTS G2_GUARDED_FILES)
	get_filename_component(guardedFile "${guardedFile}" REALPATH)
	if(guardedFile STREQUAL G2_UNMEASURED_DECLARATION_SITE)
		continue()
	endif()
	file(READ ${guardedFile} guardedText)
	foreach(unmeasuredSymbol IN LISTS G2_UNMEASURED_SYMBOLS)
		string(FIND "${guardedText}" "${unmeasuredSymbol}" symbolPosition)
		if(NOT symbolPosition EQUAL -1)
			message(FATAL_ERROR
				"${guardedFile} uses ${unmeasuredSymbol}. "
				"That value is UNMEASURED and it reads 0u. "
				"No source may use G2_MCU_BUS_CLOCK_HZ or G2_MCU_BUS_DIVIDER "
				"until it has been measured.")
		endif()
	endforeach()
endforeach()
