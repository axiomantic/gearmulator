cmake_minimum_required(VERSION 3.15)

project(g2Lib)

# THIS FILE HAS ONE OWNER: task BRD-0. Task BRD-23 is the one declared second
# writer, and it changes exactly one line -- the default of the option
# G2_LINK_MCF5307, from OFF to ON. See plan section 7.4.2.
#
# NO OTHER TASK EDITS THIS FILE. A track that needs a change here -- a new link
# library, a new compile option -- reports it to the BRD-0 owner.
#
# A track adds its own source names to its own per-track list below and to
# nothing else.

add_library(g2Lib STATIC)

# ----------------- the ten per-track source lists
#
# Each list appends to G2LIB_SOURCES. Every list is created empty by BRD-0 and
# is owned by the track its name carries.

set(G2LIB_SOURCES)

include(sources_board.cmake)
include(sources_sched.cmake)
include(sources_chain.cmake)
include(sources_dsp.cmake)
include(sources_repo.cmake)
include(sources_plugin.cmake)
include(sources_proto.cmake)
include(sources_oracle.cmake)
include(sources_perf.cmake)
include(sources_int.cmake)

# BRD-0 creates every list empty, and a CMake target with no source at all
# fails at generate time. While every list is still empty the target carries one
# translation unit GENERATED INTO THE BUILD TREE. It is not a committed source
# file: BRD-0 creates CMake files and no C++ source. The placeholder disappears
# by itself as soon as the first track adds a name to its own list.
if(NOT G2LIB_SOURCES)
	set(G2LIB_PLACEHOLDER ${CMAKE_CURRENT_BINARY_DIR}/g2LibPlaceholder.cpp)
	file(WRITE ${G2LIB_PLACEHOLDER}
		"// Generated by g2Lib/CMakeLists.txt while every per-track source list is empty.\n"
		"namespace g2 { extern const int g2LibPlaceholder; const int g2LibPlaceholder = 0; }\n")
	list(APPEND G2LIB_SOURCES ${G2LIB_PLACEHOLDER})
endif()

target_sources(g2Lib PRIVATE ${G2LIB_SOURCES})
source_group("source" FILES ${G2LIB_SOURCES})

# ----------------- the link line
#
# hardwareLib is the right layer and it carries what the G2 needs: am29f for the
# flash, i2c and i2cFlash, lcd and lcdfonts for the panel display, and sciMidi.
#
# wLib is the Waldorf layer and this target does NOT link it. See
# docs/divergence.md for the reason and the condition that would re-open it.

target_link_libraries(g2Lib PUBLIC hardwareLib dsp56kEmu synthLib)

if(DSP56300_DEBUGGER)
	target_link_libraries(g2Lib PUBLIC dsp56kDebugger)
endif()

set_property(TARGET g2Lib PROPERTY FOLDER "G2")

target_include_directories(g2Lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_include_directories(g2Lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# ----------------- the MCF5307 core
#
# The link is BEHIND AN OPTION and the option defaults to OFF, because the
# target mcf5307::mcf5307 is exported by task CPU-1 in wave 3a while this file
# is created in wave 2. Task BRD-23 turns the option ON once the target exists.
# The root CMakeLists.txt declares where the sources come from.

option(G2_LINK_MCF5307 "Link the MCF5307 core" OFF)

if(G2_LINK_MCF5307)
	if(G2_MCF5307_SOURCE_DIR)
		add_subdirectory(${G2_MCF5307_SOURCE_DIR} ${CMAKE_BINARY_DIR}/mcf5307 EXCLUDE_FROM_ALL)
	else()
		include(FetchContent)
		FetchContent_MakeAvailable(mcf5307)
	endif()
	target_link_libraries(g2Lib PUBLIC mcf5307::mcf5307)
endif()

# ----------------- the configure-time guard of measurement register rows 5 and 6
#
# G2_MCU_BUS_CLOCK_HZ and G2_MCU_BUS_DIVIDER are UNMEASURED. Spike criterion (j)
# owns both and both read 0u today. A source that uses either one computes with
# a zero, so the build fails at configure time and names the symbol.
#
# The one file allowed to carry the two names is their declaration site,
# g2Lib/g2/timebase.h, which task SCH-0 owns. Task SCH-3 drives the negative
# case that proves this guard fires.

set(G2_UNMEASURED_SYMBOLS G2_MCU_BUS_CLOCK_HZ G2_MCU_BUS_DIVIDER)

# BOTH SIDES OF THE COMPARISON ARE NORMALISED, and that is load-bearing.
# file(GLOB_RECURSE) returns each path with the glob pattern's own spelling
# preserved, so a pattern that carries ".." yields ".../g2Lib/../g2Lib/g2/
# timebase.h". That string never compares equal to the declaration site written
# without the "..", the exemption below never fires for any file, and the guard
# then rejects its own declaration site. REALPATH collapses both spellings.
get_filename_component(G2_UNMEASURED_DECLARATION_SITE
	"${CMAKE_CURRENT_SOURCE_DIR}/g2/timebase.h" REALPATH)

file(GLOB_RECURSE G2_GUARDED_FILES
	${CMAKE_CURRENT_SOURCE_DIR}/../*.c
	${CMAKE_CURRENT_SOURCE_DIR}/../*.cpp
	${CMAKE_CURRENT_SOURCE_DIR}/../*.h
	${CMAKE_CURRENT_SOURCE_DIR}/../*.hpp)

foreach(guardedFile IN LISTS G2_GUARDED_FILES)
	get_filename_component(guardedFile "${guardedFile}" REALPATH)
	if(guardedFile STREQUAL G2_UNMEASURED_DECLARATION_SITE)
		continue()
	endif()
	file(READ ${guardedFile} guardedText)
	foreach(unmeasuredSymbol IN LISTS G2_UNMEASURED_SYMBOLS)
		string(FIND "${guardedText}" "${unmeasuredSymbol}" symbolPosition)
		if(NOT symbolPosition EQUAL -1)
			message(FATAL_ERROR
				"${guardedFile} uses ${unmeasuredSymbol}. "
				"That value is UNMEASURED, it reads 0u, and spike criterion (j) owns it. "
				"Measurement register row 5 governs G2_MCU_BUS_CLOCK_HZ and row 6 governs "
				"G2_MCU_BUS_DIVIDER. No source may use either until criterion (j) reports.")
		endif()
	endforeach()
endforeach()
